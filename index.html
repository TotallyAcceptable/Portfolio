Source Architecture

Examining the blueprint for the decoupled UI Event Subsystem. Utilizing Unreal's UGameInstanceSubsystem to globally route application states without inter-actor dependencies.

UIEventSubsystem.h — C++ Header
// Architecture: Decoupled UI Subsystem
#pragma once

#include "CoreMinimal.h"
#include "Subsystems/GameInstanceSubsystem.h"
#include "UIEventSubsystem.generated.h"

UENUM(BlueprintType)
enum class EUIState : uint8 {
    MainMenu,
    Gameplay,
    Pause,
    Inventory,
    Dialog
};

DECLARE_DYNAMIC_MULTICAST_DELEGATE_OneParam(FOnUIStateRequested, EUIState, NewState);
DECLARE_DYNAMIC_MULTICAST_DELEGATE(FOnUIGoBackRequested);

UCLASS()
class UUIEventSubsystem : public UGameInstanceSubsystem {
    GENERATED_BODY()

public:
    UPROPERTY(BlueprintAssignable)
    FOnUIStateRequested OnUIStateRequested;

    UFUNCTION(BlueprintCallable, Category = "UI Interface")
    void TransitionState(EUIState TargetState) {
        ActiveState = TargetState;
        OnUIStateRequested.Broadcast(TargetState);
    }

private:
    EUIState ActiveState = EUIState::MainMenu;
};
